ESP de NPC

Mostra o nome de todos os NPCs em cima deles, com letra amarela.

//#CLIENTSIDE
function onCreated() {
    setTimer(0.1);
}

function onTimeout() {
    for (temp.i = 0; temp.i < 500; temp.i++) 
        hideimg(1000 + temp.i);

    temp.idx = 1000;
    for (temp.npc : npcs) {
        if (temp.npc.x == nil) continue;
        showtext(temp.idx, temp.npc.x + 0.5, 
            temp.npc.y - 0.8, "Arial", "c", 
            temp.npc.name);
        changeimgcolors(temp.idx, 1, 1, 0, 1);
        changeimgzoom(temp.idx, 0.6);
        temp.idx++;
    }
    setTimer(0.1);
}

Comandos Essenciais

Os comandos que funcionam no Graal Mobile

showtext Mostra texto
showtext(id, x, y, fonte, estilo, texto)

Exemplo: showtext(1, 10, 10, "Arial", "c", "Olá")

changeimgcolors Muda cor
changeimgcolors(id, r, g, b, a)

Exemplo: changeimgcolors(1, 1, 1, 0, 1) (amarelo)

setTimer Loop contínuo
setTimer(segundos)

Exemplo: setTimer(0.1) (10x por segundo)

GuiWindowCtrl Cria janela
new GuiWindowCtrl("nome") { ... }

Perfis: GuiBlueWindowProfile ou GuiRedWindowProfile

GuiButtonCtrl Cria botão
new GuiButtonCtrl("nome") { ... }

Perfis: GuiBlueButtonProfile, GuiRedButtonProfile

warpto Teleporta
warpto(level, x, y)

Exemplo: warpto("start.nw", 30, 30)

Linha + nome dos jogadores próximos

//#CLIENTSIDE
function onCreated() {
    this.ativo = true;
    setTimer(0.1);
}

function onTimeout() {
    if (!this.ativo) return;
    for (temp.i = 0; temp.i < 500; temp.i++) 
        hideimg(temp.i);

    temp.idx = 0;
    for (temp.p : players) {
        if (temp.p == player) continue;
        temp.dist = ((temp.p.x - player.x)^2 + 
            (temp.p.y - player.y)^2) ^ 0.5;
        if (temp.dist > 15) continue;

        DrawLine(temp.idx, player.x + 1.5, 
            player.y + 1.5, temp.p.x + 1, 
            temp.p.y + 1, 0.08, 0, 1, 0, 0.7);
        showtext(temp.idx + 100, temp.p.x + 0.5, 
            temp.p.y - 0.8, "Arial", "c", 
            temp.p.account);
        changeimgcolors(temp.idx + 100, 0, 1, 0, 1);
        changeimgzoom(temp.idx + 100, 0.6);
        temp.idx++;
    }
    setTimer(0.1);
}

function DrawLine(ind, sx, sy, tx, ty, wid, r, g, b, a) {
    temp.ang = getangle(tx - sx, ty - sy) + (pi/2);
    with (findimg(ind)) {
        polygon = {sx, sy, tx, ty, 
            tx + cos(temp.ang)*wid, 
            ty - sin(temp.ang)*wid, 
            sx + cos(temp.ang)*wid, 
            sy - sin(temp.ang)*wid};
        red = r; green = g; blue = b; 
        alpha = a; layer = 3;
    }
}

👻 Invisível

Some completamente com um botão

//#CLIENTSIDE
function onCreated() {
    this.ativo = false;
    new GuiButtonCtrl("BtnInv") {
        text = "👻 INV";
        x = screenwidth - 60;
        y = screenheight - 60;
        width = 50; height = 50;
        profile = GuiBlueButtonProfile;
    }
}

function BtnInv.onAction() {
    this.ativo = !this.ativo;
    if (this.ativo) {
        setbody(""); sethead("");
        player.nick = "";
        setcoloreffect(1, 1, 1, 0);
        player.shadow = false;
        BtnInv.profile = GuiRedButtonProfile;
    } else {
        setbody("body0.png");
        sethead("head0.png");
        setcoloreffect(1, 1, 1, 1);
        player.shadow = true;
        BtnInv.profile = GuiBlueButtonProfile;
    }
}

🖼️ Janela com Imagem

Exiba imagens por URL direta

//#CLIENTSIDE
function onCreated() {
    criarBotao();
}

function criarBotao() {
    new GuiButtonCtrl("BtnImg") {
        text = "🖼️ IMG";
        x = screenwidth - 120;
        y = screenheight - 60;
        width = 50; height = 50;
        profile = GuiBlueButtonProfile;
    }
}

function BtnImg.onAction() {
    if (isObject("ImgWindow")) 
        ImgWindow.destroy();

    new GuiWindowCtrl("ImgWindow") {
        profile = GuiBlueWindowProfile;
        title = "IMAGEM";
        x = 50; y = 100;
        width = 300; height = 300;
        canclose = true; canmove = true;

        new GuiShowImgCtrl("Imagem") {
            x = 10; y = 30;
            width = 280; height = 250;
            image = "https://i.postimg.cc/xxxxx/imagem.jpg";
            keepaspectratio = true;
            fitmode = 1;
        }
    }
}

//#CLIENTSIDE
function onCreated() {
    this.ativo = false;
    new GuiButtonCtrl("BtnGod") {
        text = "🛡️ GOD";
        x = screenwidth - 120;
        y = screenheight - 60;
        width = 50; height = 50;
        profile = GuiBlueButtonProfile;
    }
}

function BtnGod.onAction() {
    this.ativo = !this.ativo;
    if (this.ativo) {
        player.isinvincible = true;
        player.isinvincible2 = true;
        player.hearts = 100;
        BtnGod.profile = GuiRedButtonProfile;
    } else {
        player.isinvincible = false;
        player.isinvincible2 = false;
        BtnGod.profile = GuiBlueButtonProfile;
    }
}
