🏃 JumpAndRun

Sistema de pulo e corrida estilo plataforma.

//#CLIENTSIDE
function onCreated() {
    this.pulando = false;
    this.velocidadeY = 0;
    setTimer(0.05);
}

function onTimeout() {
    if (keydown(0) && !this.pulando) {
        this.pulando = true;
        this.velocidadeY = -0.5;
    }
    if (this.pulando) {
        player.y += this.velocidadeY;
        this.velocidadeY += 0.05;
        if (player.y >= 30) {
            player.y = 30;
            this.pulando = false;
            this.velocidadeY = 0;
        }
    }
    if (keydown(3)) player.x += 0.2;
    if (keydown(1)) player.x -= 0.2;
    setTimer(0.05);
}

Hack

⚔️ Auto Atacar

Persegue e ataca automaticamente.

//#CLIENTSIDE
function onCreated() {
    this.autoAtk = false;
    this.alvo = NULL;
    setTimer(0.05);
}

function onTimeout() {
    if (!this.autoAtk) return;
    // Procura o player mais próximo
    temp.distMin = 999;
    for (temp.p : players) {
        if (temp.p == player) continue;
        temp.dist = ((temp.p.x - player.x)^2 + (temp.p.y - player.y)^2) ^ 0.5;
        if (temp.dist < temp.distMin && temp.dist < 10) {
            temp.distMin = temp.dist;
            this.alvo = temp.p;
        }
    }
    // Persegue o alvo
    if (this.alvo != NULL) {
        temp.dx = this.alvo.x - player.x;
        temp.dy = this.alvo.y - player.y;
        if (abs(temp.dx) > 0.3) player.x += (temp.dx > 0 ? 0.2 : -0.2);
        if (abs(temp.dy) > 0.3) player.y += (temp.dy > 0 ? 0.2 : -0.2);
        // Ataca se estiver perto
        if (abs(temp.dx) < 1.5 && abs(temp.dy) < 1.5) {
            setani("sword", "");
            this.alvo = NULL;
        }
    }
    setTimer(0.05);
}

👗 Detector de Roupas

Detecta e copia roupas de outros jogadores.

//#CLIENTSIDE
function onCreated() {
    new GuiButtonCtrl("BtnDetectarRoupa") {
        text = "👗 Copiar Roupa";
        x = screenwidth - 130;
        y = screenheight - 250;
        width = 110;
        height = 40;
        profile = GuiBlueButtonProfile;
    }
}

function BtnDetectarRoupa.onAction() {
    // Detecta o jogador mais próximo
    temp.alvo = NULL;
    temp.distMin = 999;
    for (temp.p : players) {
        if (temp.p == player) continue;
        temp.dist = ((temp.p.x - player.x)^2 + (temp.p.y - player.y)^2) ^ 0.5;
        if (temp.dist < temp.distMin) {
            temp.distMin = temp.dist;
            temp.alvo = temp.p;
        }
    }
    if (temp.alvo != NULL) {
        player.body = temp.alvo.body;
        player.head = temp.alvo.head;
        player.shieldimg = temp.alvo.shieldimg;
        player.attr[1] = temp.alvo.attr[1]; // Hat
        player.attr[4] = temp.alvo.attr[4]; // Accessory
        player.chat = "Roupa copiada de " @ temp.alvo.account;
    }
}

⌨️ TypeWriter Chat

Efeito de digitação letra por letra.

//#CLIENTSIDE
function onCreated() {
    this.fraseCompleta = "";
    this.fraseAtual = "";
    this.indiceLetra = 0;
    this.velocidade = 0.08;
    this.autoAlterando = false;
}

function onPlayerChats() {
    if (this.autoAlterando) return;
    this.fraseCompleta = player.chat;
    this.fraseAtual = "";
    this.indiceLetra = 1;
    this.autoAlterando = true;
    this.fraseAtual = this.fraseCompleta.substring(0, this.indiceLetra);
    player.chat = this.fraseAtual;
    setTimer(this.velocidade);
}

function onTimeout() {
    if (this.indiceLetra >= this.fraseCompleta.length()) {
        this.autoAlterando = false;
        return;
    }
    this.indiceLetra++;
    this.fraseAtual = this.fraseCompleta.substring(0, this.indiceLetra);
    player.chat = this.fraseAtual;
    setTimer(this.velocidade);
}
